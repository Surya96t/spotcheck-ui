## Usage

```tsx
import { DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger } from "@/components/spotcheck/dropdown-menu"
import { Button } from "@/components/spotcheck/button"

<DropdownMenu>
  <DropdownMenuTrigger render={<Button />}>Actions</DropdownMenuTrigger>
  <DropdownMenuContent>
    <DropdownMenuItem onClick={rerun}>Re-run</DropdownMenuItem>
  </DropdownMenuContent>
</DropdownMenu>
```
