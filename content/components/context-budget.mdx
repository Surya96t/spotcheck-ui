## Usage

```tsx
import { ContextBudget } from "@/components/spotcheck/context-budget"

<ContextBudget
  limit={200000}
  segments={[
    { label: "System", tokens: 3200 },
    { label: "Tools", tokens: 9800 },
    { label: "History", tokens: 41200 },
    { label: "Files", tokens: 62400 },
  ]}
  onCompact={compact}
/>
```
