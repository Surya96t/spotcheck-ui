## Usage

```tsx
"use client"

import { DetentSlider } from "@/components/spotcheck/detent-slider"

const zones = [
  { from: 0, to: 0.4, label: "Deterministic" },
  { from: 0.6, to: 1.2, label: "Balanced" },
  { from: 1.4, to: 2, label: "Creative" },
]

<DetentSlider
  label="Temperature"
  min={0}
  max={2}
  step={0.2}
  defaultValue={0.8}
  zones={zones}
  onValueChange={setTemperature}
/>
```

`onValueChange` runs when the slider lands on a step. Screen readers hear the value and its zone
together, such as "0.8, Balanced".
