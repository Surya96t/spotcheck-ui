## Usage

```tsx
import { Popover, PopoverContent, PopoverTrigger } from "@/components/spotcheck/popover"
import { Button } from "@/components/spotcheck/button"

<Popover>
  <PopoverTrigger render={<Button />}>Details</PopoverTrigger>
  <PopoverContent>…</PopoverContent>
</Popover>
```
