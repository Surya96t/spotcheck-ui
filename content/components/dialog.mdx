## Usage

```tsx
import { Dialog, DialogClose, DialogContent, DialogTrigger } from "@/components/ledger/dialog"
import { Button } from "@/components/ledger/button"

<Dialog>
  <DialogTrigger render={<Button variant="ink" />}>New key</DialogTrigger>
  <DialogContent title="New API key" description="It's shown once.">
    <DialogClose render={<Button />}>Done</DialogClose>
  </DialogContent>
</Dialog>
```
